       ```typescript
       /**
        * Detect HTTP 409 Conflict errors from the API.
        * Used for IPNS publish conflict detection (sequence number mismatch).
        */
       export function isConflictError(error: unknown): boolean {
         if (!error || typeof error !== 'object') return false;
         // Orval-generated client throws errors with status property
         const e = error as Record<string, unknown>;
         if (e.status === 409) return true;
         // Fallback: check nested response
         if (typeof e.response === 'object' && e.response !== null) {
           const resp = e.response as Record<string, unknown>;
           if (resp.status === 409) return true;
         }
         return false;
       }

       /**
        * Extract the current server sequence number from a 409 conflict error.
        */
       export function getConflictSequenceNumber(error: unknown): string | undefined {
         if (!error || typeof error !== 'object') return undefined;
         const e = error as Record<string, unknown>;
         // Check for currentSequenceNumber in error body/data
         if (typeof e.data === 'object' && e.data !== null) {
           return (e.data as Record<string, unknown>).currentSequenceNumber as string | undefined;
         }
         return undefined;
       }
       ```

       **IMPORTANT:** Check how Orval-generated client surfaces HTTP errors. Look at the generated `apps/web/src/api/ipns/ipns.ts` to see the error shape. The `isConflictError` function must match the actual error structure. The Orval custom instance (`apps/web/src/api/custom-instance.ts` or similar) likely wraps errors. Inspect that file and adjust the detection logic accordingly.

    **2. Hooks layer -- handle 409 with re-sync + retry:**

    In `useFolderMutations.ts`, create a shared conflict handler:
    ```typescript
    import { isConflictError } from '../lib/errors';
    import { useSyncStore } from '../stores/sync.store';

    // Inside the hook, create a reusable conflict handler:
    const handleConflictRetry = useCallback(async (
      operation: () => Promise<void>,
      folderIpnsName: string,
    ) => {
      try {
        await operation();
      } catch (err) {
        if (!isConflictError(err)) throw err;

        // 1. Show conflict status
        useSyncStore.getState().setConflict('Folder updated by another device, re-syncing...');

        // 2. Re-sync: resolve IPNS, fetch & decrypt metadata, update store
        // Use the sync callback from FileBrowser (or directly call resolveIpnsRecord + loadFolder)
        const resolved = await resolveIpnsRecord(folderIpnsName);
        if (resolved) {
          // Reload folder from fresh IPNS data
          // Get fresh folder state from store
          const freshState = useFolderStore.getState();
          // The sync will update the store with new children and sequenceNumber
          // After sync, the store has the latest remote state
        }

        // 3. Add random jitter (100-500ms) before retry to break symmetry
        await new Promise(r => setTimeout(r, 100 + Math.random() * 400));

        // 4. Retry ONCE with fresh state
        try {
          await operation();
          useSyncStore.getState().clearConflict();
        } catch (retryErr) {
          useSyncStore.getState().clearConflict();
          if (isConflictError(retryErr)) {
            throw new Error('Conflict persists after re-sync. Please try again.');
          }
          throw retryErr;
        }
      }
    }, []);
    ```

    **CRITICAL -- Stale closure prevention:** The `operation` callback MUST re-read folder state from `useFolderStore.getState()` (not from the closure). Each mutation handler (handleCreate, handleRename, handleMove, handleDelete, handleDeleteItems) should be wrapped so that:
    - The first attempt uses the current state from the hook
    - The retry re-reads fresh state from `useFolderStore.getState()` after re-sync

    The exact implementation approach: In each mutation handler that calls `updateFolderMetadata` or batch publish, wrap the call in a try/catch for conflict errors. On conflict:
    1. Call sync to refresh the store
    2. Re-read parent folder from `useFolderStore.getState()`
    3. Rebuild the operation with fresh children/sequenceNumber
    4. Retry once

    **Apply to these mutation handlers:**
    - `handleCreate` (creates subfolder)
    - `handleRename` (renames item)
    - `handleMove` / `handleMoveItems` (move item between folders)
    - `handleDelete` / `handleDeleteItems` (delete item)

    In `useFileOperations.ts`:
    - `handleAddFile` / `handleAddFiles`: These use `addFileToFolder`/`addFilesToFolder` which do batch publish. Wrap with similar conflict handling. On conflict, re-sync parent folder, rebuild the batch with fresh children list, retry.
    - `handleUpdateFile`: This publishes only the file IPNS record (not folder). NO conflict detection needed here.
