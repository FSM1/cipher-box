   ```typescript
   /**
    * Prometheus HTTP metrics middleware for Express.
    * Mirrors the API's cipherbox_http_request_duration_seconds pattern
    * with a tee-specific prefix for Grafana dashboard compatibility.
    */
   import { Histogram, Counter, register } from 'prom-client';

   const httpDuration = new Histogram({
     name: 'cipherbox_tee_http_request_duration_seconds',
     help: 'TEE worker HTTP request duration in seconds',
     labelNames: ['method', 'route', 'status_code'],
     buckets: [0.01, 0.05, 0.1, 0.25, 0.5, 1, 2.5, 5],
   });

   const republishEntries = new Counter({
     name: 'cipherbox_tee_republish_entries_total',
     help: 'Total IPNS entries processed by TEE republish',
     labelNames: ['result'], // 'success' | 'failure'
   });

   const migrationCids = new Counter({
     name: 'cipherbox_tee_migration_cids_total',
     help: 'Total CIDs processed by TEE migration',
     labelNames: ['result'], // 'success' | 'failure'
   });

   export { republishEntries, migrationCids };

   export function metricsMiddleware(req, res, next) {
     if (req.path === '/metrics') return next(); // Don't instrument self
     const start = process.hrtime.bigint();
     res.on('finish', () => {
       const durationNs = Number(process.hrtime.bigint() - start);
       httpDuration.observe(
         { method: req.method, route: req.route?.path ?? req.path, status_code: res.statusCode },
         durationNs / 1e9
       );
     });
     next();
   }
   ```

   Use the API's naming convention (`cipherbox_tee_*` prefix) so both the API and TEE worker metrics can coexist on the same Grafana dashboard.

1. Create `apps/tee-worker/src/services/logger.ts`:

   ```typescript
   /**
    * Minimal structured JSON logger for TEE worker.
    * No external dependencies -- outputs newline-delimited JSON to stdout/stderr.
    * Compatible with Grafana Loki, CloudWatch, and other log aggregators.
    */
   type LogLevel = 'info' | 'warn' | 'error';

   function log(level: LogLevel, message: string, data?: Record<string, unknown>) {
     const entry = {
       timestamp: new Date().toISOString(),
       level,
       service: 'tee-worker',
       message,
       ...data,
     };
     const output = JSON.stringify(entry);
     if (level === 'error') {
       process.stderr.write(output + '\n');
     } else {
       process.stdout.write(output + '\n');
     }
   }

   export const logger = {
     info: (message: string, data?: Record<string, unknown>) => log('info', message, data),
     warn: (message: string, data?: Record<string, unknown>) => log('warn', message, data),
     error: (message: string, data?: Record<string, unknown>) => log('error', message, data),
   };
   ```
