1. The existing `build-tee` job in deploy-staging.yml builds and pushes the TEE Docker image to GHCR. This job stays as-is -- Phala Cloud pulls the image from GHCR.

2. Add a new job `deploy-tee-phala` that runs after `build-tee` and deploys the CVM to Phala Cloud:

   ```yaml
   deploy-tee-phala:
     name: Deploy TEE Worker to Phala Cloud
     needs: [build-tee]
     runs-on: ubuntu-latest
     environment: staging
     steps:
       - uses: actions/checkout@v4
         with:
           ref: ${{ inputs.staging_tag || github.ref_name }}

       - uses: actions/setup-node@v4
         with:
           node-version: '22'

       - name: Install Phala CLI
         run: npm install -g phala

       - name: Deploy CVM to Phala Cloud
         env:
           PHALA_CLOUD_API_KEY: ${{ secrets.PHALA_CLOUD_API_KEY }}
           TEE_WORKER_SECRET: ${{ secrets.STAGING_TEE_WORKER_SECRET }}
         run: |
           set -euo pipefail
           OWNER="${{ github.repository_owner }}"
           export GITHUB_REPOSITORY_OWNER="${OWNER,,}"
           export TAG="${{ env.DEPLOY_TAG }}"

           # Substitute env vars in compose file
           envsubst < tee-worker/docker-compose.phala.yml > /tmp/docker-compose.phala.yml

           # Deploy or update CVM (same name = update, preserving app_id and derived keys)
           phala deploy \
             -c /tmp/docker-compose.phala.yml \
             -n cipherbox-tee-staging \
             -e "TEE_WORKER_SECRET=${TEE_WORKER_SECRET}" \
             --wait

           # Output CVM info for debugging
           phala cvms get cipherbox-tee-staging || true
   ```
