```javascript
// Vault blob v2 parsing (inline in recovery.html)
const BLOB_V2_VERSION = 0x02;

function detectBlobVersion(blob) {
  return (blob.length > 0 && blob[0] === BLOB_V2_VERSION) ? 2 : 1;
}

function deserializeVaultBlobV2(blob) {
  if (blob.length < 3) throw new Error('Blob too short for v2 header');
  if (blob[0] !== BLOB_V2_VERSION) throw new Error('Not a v2 vault blob');
  const keyLen = (blob[1] << 8) | blob[2];
  if (blob.length < 3 + keyLen) throw new Error('Blob too short for key');
  return {
    encryptedRootFolderKey: blob.slice(3, 3 + keyLen),
    encryptedMetadataJson: blob.slice(3 + keyLen),
  };
}
```
