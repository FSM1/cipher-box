```typescript
/** Result of a pin operation */
export type PinResult = {
  cid: string;
  size: number;
};

/** Status of a pin */
export type PinStatus = {
  cid: string;
  status: 'queued' | 'pinning' | 'pinned' | 'failed';
};

/** Abstract pinning provider -- implemented by KuboProvider and PsaProvider */
export interface PinningProvider {
  /** Upload and pin data, returning CID and size */
  pin(data: Uint8Array, name?: string): Promise<PinResult>;
  /** Remove a pin by CID */
  unpin(cid: string): Promise<void>;
  /** Check pin status by CID */
  status(cid: string): Promise<PinStatus>;
  /** Fetch pinned content by CID */
  get(cid: string): Promise<Uint8Array>;
}

/** User-selectable pinning mode */
export type PinningMode = 'cipherbox' | 'external' | 'dual';

/** Configuration for an external IPFS provider */
export type ExternalProviderConfig = {
  endpoint: string;
  authToken: string;
  protocol: 'psa' | 'kubo';
  providerName?: string;
};

/** Result of a connection test */
export type ConnectionTestResult = {
  success: boolean;
  protocol?: 'kubo' | 'psa';
  version?: string;
  latencyMs: number;
  error?: string;
  corsError?: boolean;
  corsInstructions?: string;
};
```

**2. Create `packages/sdk-core/src/pinning/kubo-provider.ts`:**

Implements `PinningProvider` using Kubo RPC API (`/api/v0/*`):

- `pin(data)`: POST `/api/v0/add?pin=true&cid-version=1` with FormData (Blob from Uint8Array -- pass typed array directly to Blob, never `.buffer`). Parse ndjson response: `{ Hash, Size }`. Return `{ cid: Hash, size: parseInt(Size, 10) }`.
- `unpin(cid)`: POST `/api/v0/pin/rm?arg=${encodeURIComponent(cid)}`. Ignore "not pinned" errors (check response text).
- `status(cid)`: POST `/api/v0/pin/ls?arg=${encodeURIComponent(cid)}`. Return `{ cid, status: 'pinned' }` on success, `{ cid, status: 'failed' }` on error.
- `get(cid)`: POST `/api/v0/cat?arg=${encodeURIComponent(cid)}`. Return `new Uint8Array(await response.arrayBuffer())`.
- **SECURITY:** All CID parameters in URL construction MUST use `encodeURIComponent(cid)` to prevent parameter injection. While CIDs are normally safe alphanumeric strings, defense-in-depth requires encoding.
- Constructor takes `endpoint: string` and optional `authToken?: string`.
- Auth header: `Authorization: Basic ${authToken}` when token provided.
- All fetch calls use `signal: AbortSignal.timeout(30_000)` for 30s timeout.
- Normalize endpoint: strip trailing slash in constructor.

**3. Create `packages/sdk-core/src/pinning/psa-provider.ts`:**

Implements `PinningProvider` for IPFS Pinning Service API:

- `pin(data)`: Throw `Error('PsaProvider.pin() cannot upload raw data. Use pinByCid() after uploading via CipherBox relay or Kubo node.')` -- PSA is CID-reference-only.
- `pinByCid(cid, name?)`: POST `/pins` with JSON body `{ cid, name: name ?? 'cipherbox-' + Date.now() }`. Header: `Authorization: Bearer ${authToken}`, `Content-Type: application/json`. Return `{ cid: result.pin.cid, status: result.status }`.
- `unpin(cid)`: First GET `/pins?cid={cid}&status=pinned,pinning,queued` to find `requestid`(s). Then DELETE `/pins/{requestid}` for each. PSA uses requestid not CID for deletion.
- `status(cid)`: GET `/pins?cid={cid}&limit=1`. Parse result count and status.
- `get(cid)`: Throw `Error('PsaProvider does not support content retrieval. Use an IPFS gateway.')`.
- Constructor takes `endpoint: string` and `authToken: string` (required for PSA).
- All fetch calls use `signal: AbortSignal.timeout(30_000)`.
- Auth header: `Authorization: Bearer ${authToken}`.
- Normalize endpoint: strip trailing slash.

**4. Create `packages/sdk-core/src/pinning/connection-test.ts`:**

Export `testConnection(endpoint: string, authToken?: string): Promise<ConnectionTestResult>`:

Sequential probe strategy:

1. Try Kubo: POST `{endpoint}/api/v0/id` with Basic auth (if token). On success, parse `{ AgentVersion }` and return `{ success: true, protocol: 'kubo', version: data.AgentVersion, latencyMs }`.
2. On TypeError (CORS): Return `{ success: false, corsError: true, corsInstructions: kuboCorsInstructions(endpoint), error: 'CORS error: browser blocked the request. configure cors on your kubo node.' }`.
3. Try PSA: GET `{endpoint}/pins?limit=1` with Bearer auth. On 200: return `{ success: true, protocol: 'psa', latencyMs }`. On 401: return `{ success: false, protocol: 'psa', error: 'authentication failed. check your auth token.' }`.
4. On TypeError (CORS for PSA): Return with corsError true and PSA CORS instructions.
5. Neither works: Return `{ success: false, error: 'could not detect ipfs protocol at this endpoint.' }`.

Timeout: `AbortSignal.timeout(10_000)` for each probe.

Helper functions (not exported):

- `kuboCorsInstructions(endpoint)`: Returns multiline string:
  ```
  ipfs config --json API.HTTPHeaders.Access-Control-Allow-Origin '["https://app.cipherbox.cc", "http://localhost:5173"]'
  ipfs config --json API.HTTPHeaders.Access-Control-Allow-Methods '["POST"]'
  ipfs config --json API.HTTPHeaders.Access-Control-Allow-Headers '["Authorization", "Content-Type"]'
  ```
- `psaCorsInstructions()`: Returns `'check your pinning service dashboard for cors/allowed-origins settings. add https://app.cipherbox.cc to the allowed origins list.'`

CORS detection: TypeError with message containing 'Failed to fetch' or 'NetworkError' or 'Network request failed' (browser-dependent). Also catch DOMException name 'AbortError' for timeouts separately.
