    Add TC06 after TC05 in the serial test suite:

    ```typescript
    // ============================================================
    // TC06: Permanent delete reclaims storage quota (GAP-1 fix)
    // ============================================================

    test('TC06: permanent delete reclaims storage quota', async () => {
      test.slow();

      const fileName = `bin-quota-${runId}.txt`;
      const fileContent = 'x'.repeat(1024); // 1KB file for measurable quota change

      // Navigate to files and upload a test file
      await navigateToFiles();
      const testFile = createTestTextFile(fileName, fileContent);
      await uploadZone.uploadFile(testFile.path);
      await fileList.waitForItemToAppear(fileName, { timeout: 60000 });

      // Read quota before delete (from the quota indicator in the UI)
      // Wait a moment for quota to update after upload
      await page.waitForTimeout(3000);
      const quotaBefore = await page.evaluate(() => {
        // Access Zustand store directly for precise quota reading
        const store = (window as any).__ZUSTAND_STORES__?.quota;
        if (store) return store.getState().usedBytes;
        // Fallback: read from DOM
        const el = document.querySelector('[data-testid="quota-used"]');
        return el ? parseInt(el.textContent || '0', 10) : -1;
      });

      // Delete the file (soft-delete -> bin)
      await fileList.rightClickItem(fileName);
      await contextMenu.waitForOpen();
      await contextMenu.clickDelete();
      await confirmDialog.waitForOpen();
      await confirmDialog.clickConfirm();
      await fileList.waitForItemToDisappear(fileName, { timeout: 30000 });

      // Navigate to bin and permanently delete
      await binPage.navigate();
      await binPage.waitForBinItem(fileName, { timeout: 30000 });
      await binPage.permanentlyDeleteItem(fileName);
      await binPage.waitForBinItemToDisappear(fileName, { timeout: 30000 });

      // Wait for quota to update after permanent delete + unpin
      await page.waitForTimeout(3000);
      const quotaAfter = await page.evaluate(() => {
        const store = (window as any).__ZUSTAND_STORES__?.quota;
        if (store) return store.getState().usedBytes;
        const el = document.querySelector('[data-testid="quota-used"]');
        return el ? parseInt(el.textContent || '0', 10) : -1;
      });

      // Quota should decrease (or at least not increase) after permanent delete
      // If quotaBefore was -1 (store not accessible), skip the assertion
      if (quotaBefore > 0 && quotaAfter >= 0) {
        expect(quotaAfter).toBeLessThan(quotaBefore);
      }
    });
    ```

    **Important considerations:**
    - The Zustand store may not be directly accessible via `window`. Check if there's an

      approach: check the quota display text in the sidebar or settings.
    - If the quota display is not in the sidebar, simplify the test to just verify that
      permanent delete completes without errors (the actual quota update is an internal detail
      that unit tests should verify). In that case, the test verifies the end-to-end flow
      works (upload -> soft-delete -> permanent delete -> no crash), which exercises the
      contentCid path.
    - Look at how `useQuotaStore` exposes quota to the UI. If there's a
      `[data-testid="quota-used"]` or similar element, use that for reading.
    - If no quota UI element exists, the test still validates that permanent delete runs the
      full unpinFileCids code path without errors (which was crashing before the GAP-1 fix
      because it tried to parse encrypted data as JSON).
