Rewrite `.github/workflows/tag-staging.yml` to use date-based staging tags per D-35.

**Key changes:**

1. **Remove release_tag input** — staging tags are no longer derived from a release tag. The workflow deploys from the current main branch HEAD.

2. **New input: optional description** — for human-readable context (optional).

3. **Tag format:** `staging-YYYYMMDD-release-N` where N is a sequential counter per date.

4. **E2E still runs against the tag/commit** — keep web-e2e and desktop-e2e verification.

Updated workflow:

```yaml
name: Tag Staging Release

on:
  workflow_dispatch:
    inputs:
      description:
        description: 'Optional description for this staging release'
        required: false
        type: string

jobs:
  web-e2e:
    name: Web E2E
    uses: ./.github/workflows/web-e2e.yml
    with:
      ref: main
    secrets: inherit

  desktop-e2e:
    name: Desktop E2E
    uses: ./.github/workflows/desktop-e2e.yml
    with:
      ref: main
    secrets: inherit

  tag-staging:
    name: Create Staging Tag
    needs: [web-e2e, desktop-e2e]
    runs-on: ubuntu-latest
    environment: staging-approval
    permissions:
      contents: write
    outputs:
      staging_tag: ${{ steps.tag.outputs.staging_tag }}
    steps:
      - uses: actions/checkout@v6
        with:
          fetch-depth: 0

      - name: Calculate staging tag
        id: tag
        run: |
          DATE=$(date -u +%Y%m%d)
          STAGING_PREFIX="staging-${DATE}-release"

          # Find existing tags for today and get the highest release number
          LAST_N=$(git tag -l "${STAGING_PREFIX}-*" | sed "s/${STAGING_PREFIX}-//" | sort -n | tail -1)
          if [ -z "$LAST_N" ]; then
            NEXT_N=1
          else
            NEXT_N=$((LAST_N + 1))
          fi

          STAGING_TAG="${STAGING_PREFIX}-${NEXT_N}"
          echo "staging_tag=${STAGING_TAG}" >> "$GITHUB_OUTPUT"

          echo "## Staging Tag" >> "$GITHUB_STEP_SUMMARY"
          echo "Creating **${STAGING_TAG}** from main HEAD" >> "$GITHUB_STEP_SUMMARY"
          if [ -n "${{ inputs.description }}" ]; then
            echo "" >> "$GITHUB_STEP_SUMMARY"
            echo "Description: ${{ inputs.description }}" >> "$GITHUB_STEP_SUMMARY"
          fi

      - name: Create and push staging tag
        env:
          STAGING_TAG: ${{ steps.tag.outputs.staging_tag }}
        run: |
          git tag "${STAGING_TAG}" HEAD
          git push origin "${STAGING_TAG}"

  deploy:
    name: Deploy Staging
    needs: tag-staging
    permissions:
      contents: write
      packages: write
    uses: ./.github/workflows/deploy-staging.yml
    with:
      staging_tag: ${{ needs.tag-staging.outputs.staging_tag }}
    secrets: inherit
```

Key differences from current:

- No `release_tag` input — always tags from main HEAD
- No `validate-tag` job (was validating the release tag existed)
- Date-based tag format: `staging-YYYYMMDD-release-N`
- E2E runs against `ref: main` (not a release tag)
- Tag created on HEAD, not on a specific release tag
- Description input for optional context
