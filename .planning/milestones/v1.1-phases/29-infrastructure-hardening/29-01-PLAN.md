# Plan 29-01: IPNS Batch Unenroll API Endpoint

<objective>
Expose the existing `RepublishService.unenrollIpns()` via a new REST endpoint in `IpnsController`. Add a batch variant following the `POST /ipns/publish-batch` pattern (max 200 IPNS names per call). Regenerate the API client so `@cipherbox/api-client` exposes the new function.
</objective>

<must_haves>

- A POST endpoint at `/ipns/unenroll` accepts a body with an array of IPNS names and deletes their TEE republish enrollments
- The endpoint requires JWT auth and scopes deletions to the authenticated user's userId
- The endpoint accepts up to 200 IPNS names per call (matching batch publish limit)
- The API client is regenerated and exports the new unenroll function
  </must_haves>

```typescript
import { ApiProperty } from '@nestjs/swagger';
import { IsArray, IsString, ArrayMaxSize, ArrayMinSize, Matches, MaxLength } from 'class-validator';

export class BatchUnenrollIpnsDto {
  @ApiProperty({
    description: 'Array of IPNS names to unenroll from TEE republishing (max 200)',
    type: [String],
    example: ['k51qzi5uqu5dkkciu33khkzbcmxtyhn2hgdqyp6rv7s5egjlsdj6a2xpz9lxvz'],
  })
  @IsArray()
  @ArrayMinSize(1)
  @ArrayMaxSize(200)
  @IsString({ each: true })
  @Matches(/^(k51qzi5uqu5[a-z0-9]{40,60}|bafzaa[a-z2-7]{50,70})$/, {
    each: true,
    message: 'Each ipnsName must be a valid CIDv1 libp2p-key (k51qzi5uqu5... or bafzaa...)',
  })
  @MaxLength(70, { each: true })
  ipnsNames!: string[];
}

export class BatchUnenrollIpnsResponseDto {
  @ApiProperty({ description: 'Number of IPNS names successfully unenrolled', example: 5 })
  totalUnenrolled!: number;

  @ApiProperty({ description: 'Total IPNS names in the request', example: 5 })
  totalRequested!: number;
}
```

3. Add an `unenrollBatch` method to `IpnsService` in `apps/api/src/ipns/ipns.service.ts`:

```typescript
async unenrollBatch(userId: string, ipnsNames: string[]): Promise<{ totalUnenrolled: number }> {
  let unenrolled = 0;
  for (const ipnsName of ipnsNames) {
    try {
      await this.republishService.unenrollIpns(userId, ipnsName);
      unenrolled++;
    } catch (err) {
      this.logger.warn(`Failed to unenroll ${ipnsName}: ${err instanceof Error ? err.message : err}`);
    }
  }
  return { totalUnenrolled: unenrolled };
}
```

4. Add endpoint to `IpnsController` in `apps/api/src/ipns/ipns.controller.ts`:

```typescript
@Throttle({ default: { limit: 5, ttl: 60000 } }) // 5 batch unenrolls per minute per user
@Post('unenroll')
@ApiOperation({
  summary: 'Batch unenroll IPNS records from TEE republishing',
  description:
    'Remove IPNS names from the TEE republish schedule. Called after file/folder deletion ' +
    'to prevent orphaned records from accumulating. Accepts up to 200 IPNS names per call.',
})
@ApiResponse({
  status: 200,
  description: 'Unenrollment results',
  type: BatchUnenrollIpnsResponseDto,
})
@ApiResponse({ status: 400, description: 'Bad Request - Invalid IPNS name format or array too large' })
@ApiResponse({ status: 401, description: 'Unauthorized - JWT token required' })
async unenrollBatch(
  @Request() req: RequestWithUser,
  @Body() dto: BatchUnenrollIpnsDto,
): Promise<BatchUnenrollIpnsResponseDto> {
  const result = await this.ipnsService.unenrollBatch(req.user.id, dto.ipnsNames);
  return {
    totalUnenrolled: result.totalUnenrolled,
    totalRequested: dto.ipnsNames.length,
  };
}
```

Import `BatchUnenrollIpnsDto` and `BatchUnenrollIpnsResponseDto` from `./dto` in the controller's import list.
