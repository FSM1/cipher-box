```typescript
/**
 * Purge expired bin entries based on retention policy.
 * Cleans up IPFS CIDs and removes entries from bin metadata.
 */
export async function purgeExpiredEntries(params: {
  binState: BinState;
  retentionDays: number;
  opCtx: BinOperationContext;
}): Promise<{ purgedCount: number; updatedState: BinState }> {
  const retentionMs = params.retentionDays * 24 * 60 * 60 * 1000;
  const now = Date.now();

  const expired = params.binState.entries.filter((e) => now - e.deletedAt > retentionMs);
  if (expired.length === 0) return { purgedCount: 0, updatedState: params.binState };

  // Cleanup CIDs for expired entries (best-effort, matches bin.service pattern)
  for (const entry of expired) {
    try {
      await cleanupEntryCids(entry, params.opCtx);
    } catch {
      // Best-effort: CID cleanup failures are non-blocking
    }
  }

  // Remove expired entries and publish updated bin
  const expiredIds = new Set(expired.map((e) => e.id));
  const remainingEntries = params.binState.entries.filter((e) => !expiredIds.has(e.id));
  const newSeq = params.binState.sequenceNumber + 1;

  const metadata: RecycleBinMetadata = {
    version: BIN_METADATA_VERSION,
    sequenceNumber: newSeq,
    entries: remainingEntries,
  };

  await saveBinMetadataInternal({ metadata, ...params.opCtx });

  return {
    purgedCount: expired.length,
    updatedState: {
      entries: remainingEntries,
      sequenceNumber: newSeq,
      ipnsName: params.binState.ipnsName,
    },
  };
}
```

Note: `cleanupEntryCids` and `saveBinMetadataInternal` are internal helpers already in the SDK bin module (check exact names — they may be `cleanupEntryCids`/`saveBinMetadata` or similar). Reuse existing internal helpers. If `cleanupEntryCids` doesn't exist in SDK bin, extract it from bin.service.ts (lines 687-693 and helper functions below it).

2. In `packages/sdk/src/client.ts`, add `purgeExpired` method to `CipherBoxClient`:

```typescript
/**
 * Purge expired entries from the recycle bin.
 * Cleans up IPFS CIDs for expired entries and removes them from bin metadata.
 *
 * @param retentionDays - Retention period in days. Entries older than this are purged.
 * @returns Number of entries purged
 */
async purgeExpired(retentionDays: number): Promise<number> {
  if (!this._binState) throw new BinNotLoadedError();

  const { purgedCount, updatedState } = await purgeExpiredEntries({
    binState: this._binState,
    retentionDays,
    opCtx: this._buildBinContext(),
  });

  if (purgedCount > 0) {
    this._binState = updatedState;
    this._emitter.emit({
      type: 'bin:updated',
      entries: updatedState.entries,
      sequenceNumber: updatedState.sequenceNumber,
    });
  }

  return purgedCount;
}
```

Note: Check the exact property names for bin state on the client (e.g., `this._binState`, `this._buildBinContext()`). Read `client.ts` to find the correct internal names and patterns. Follow the same pattern as `emptyBin()` or `permanentDelete()`.

- Remove: `import { initializeBin, purgeExpired } from '../services/bin.service';`
- The `loadBin` callback currently does:
  1. `await initializeBin(...)` — populates useBinStore from IPNS
  2. `await getSdkClient().loadBin()` — also loads into SDK state

Replace with:

1. `const binState = await getSdkClient().loadBin()` — SDK loads from IPNS, returns BinState
2. Update useBinStore from the returned binState:
   ```typescript
   const binStore = useBinStore.getState();
   binStore.setBinIpnsName(binState.ipnsName);
   binStore.setEntries(binState.entries, binState.sequenceNumber);
   ```
3. `await getSdkClient().purgeExpired(retentionDays)` — SDK handles purge
4. After purge, re-read SDK state to update store (via bin:updated event subscription, or read directly)

The `loadBin` function should catch errors gracefully (non-blocking, same as current behavior). Keep the `setState` loading/error pattern.

Also update the `purge` callback to use `getSdkClient().purgeExpired(retentionDays)` instead of the service function.

- Remove: `import { initializeBin } from '../services/bin.service';`

- Note: The SDK client must be initialized before calling loadBin. Check the call ordering: `initSdkClient(...)` should come before the bin init call. If `initializeBin` is called before SDK init, the ordering needs adjustment — the SDK must be initialized first.
- Check if `useBinStore` is updated via an event subscription from the SDK client (if so, no manual store update needed). If not, add manual store update after `loadBin()`.

### Verify event subscriptions

Check if there's an existing SDK event subscription that bridges `bin:updated` events to `useBinStore`. Look in `apps/web/src/` for `bin:updated` or SDK event listener setup. If it exists, no manual store update is needed after SDK operations. If not, consider adding one (or do manual updates in the hooks).
