**4. Update `apps/desktop/src-tauri/src/sync/mod.rs`:**

This becomes a thin wrapper:

```rust
//! Sync daemon bridge for desktop app.
//! Creates cipherbox_sdk::SyncDaemon with Tauri tray status callback.

use cipherbox_sdk::{SyncDaemon, SyncStatus};
use crate::tray;

pub async fn start_sync_daemon(
    state: &crate::state::AppState,
    app_handle: tauri::AppHandle,
) -> SyncDaemon {
    let app = app_handle.clone();
    let status_callback = Arc::new(move |status: SyncStatus| {
        // Bridge SDK status to Tauri tray update
        let tray_status = match status {
            SyncStatus::Idle => tray::TrayStatus::Idle,
            SyncStatus::Syncing => tray::TrayStatus::Syncing,
            SyncStatus::Error(e) => tray::TrayStatus::Error(e),
        };
        tray::update_tray_status(&app, tray_status);
    });

    SyncDaemon::new(state.sdk.clone(), status_callback, Duration::from_secs(30))
}
```

Remove `queue.rs` module declaration from `sync/mod.rs` (it's in the crate now).
