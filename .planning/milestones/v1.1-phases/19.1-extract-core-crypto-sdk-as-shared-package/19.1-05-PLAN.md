<objective>
Rewire the web app's React hooks and Zustand stores to use the CipherBoxClient SDK instead of calling service functions directly. Hooks become thin wrappers, stores become SDK event subscribers.

Purpose: Complete the migration from "services + hooks + stores" architecture to "SDK + thin hooks + event-driven stores." This is the consumer-side integration that makes the SDK the single source of truth for all file operations.

Output: Web app fully integrated with @cipherbox/sdk. All hooks refactored. All stores subscribing to SDK events. Build passes. Human verification confirms no regressions.
</objective>

       ```typescript
       import { CipherBoxClient, type CipherBoxClientConfig } from '@cipherbox/sdk';

       let _client: CipherBoxClient | null = null;

       /**
        * Initialize the SDK client. Called after vault is loaded (login complete).
        * The client persists for the session and is destroyed on logout.
        */
       export function initSdkClient(config: CipherBoxClientConfig): CipherBoxClient {
         if (_client) {
           _client.destroy();
         }
         _client = new CipherBoxClient(config);
         return _client;
       }

       /**
        * Get the current SDK client instance.
        * Throws if called before initSdkClient() (i.e., before login).
        */
       export function getSdkClient(): CipherBoxClient {
         if (!_client) {
           throw new Error('SDK client not initialized. Call initSdkClient() after login.');
         }
         return _client;
       }

       /**
        * Check if the SDK client is initialized (for conditional usage).
        */
       export function hasSdkClient(): boolean {
         return _client !== null;
       }

       /**
        * Destroy the SDK client. Called on logout.
        * Clears internal state, key caches, and event subscriptions.
        */
       export function destroySdkClient(): void {
         if (_client) {
           _client.destroy();
           _client = null;
         }
       }
       ```

       ```typescript
       import { getSdkClient } from '../lib/sdk-provider';
       import { useFolderStore } from '../stores/folder.store';
       import { useNotification } from '../stores/notification.store';

       export function useFolderMutations() {
         const { addToast } = useNotification();
         const currentFolderIpnsName = useFolderStore((s) => s.currentFolderIpnsName);

         const createFolder = useCallback(async (name: string) => {
           try {
             const client = getSdkClient();
             await client.createFolder(currentFolderIpnsName, name);
             // SDK emits folder:updated event -> store subscribes -> UI updates
           } catch (error) {
             addToast({ type: 'error', message: `Failed to create folder: ${(error as Error).message}` });
             throw error;
           }
         }, [currentFolderIpnsName, addToast]);

         const renameItem = useCallback(async (childId: string, newName: string) => {
           try {
             const client = getSdkClient();
             await client.renameItem(currentFolderIpnsName, childId, newName);
       ```
