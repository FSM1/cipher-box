    4. Create `packages/sdk/src/events.ts`:
       ```typescript
       import type { FolderChild } from '@cipherbox/core';

       export type SdkEvent =
         | { type: 'folder:loaded'; folderId: string; ipnsName: string; children: FolderChild[] }
         | { type: 'folder:updated'; folderId: string; ipnsName: string; children: FolderChild[] }
         | { type: 'folder:deleted'; folderId: string }
         | { type: 'file:uploaded'; folderId: string; fileName: string; cid: string }
         | { type: 'file:downloaded'; cid: string }
         | { type: 'bin:updated'; entries: unknown[] }
         | { type: 'operation:start'; operation: string }
         | { type: 'operation:end'; operation: string; durationMs: number }
         | { type: 'error'; operation: string; error: Error };

       export type SdkEventHandler = (event: SdkEvent) => void;

       /**
        * Simple typed event emitter. No external dependencies.
        * Consumers subscribe with on() and unsubscribe with off().
        */
       export class SdkEventEmitter {
         private handlers: SdkEventHandler[] = [];

         on(handler: SdkEventHandler): () => void {
           this.handlers.push(handler);
           return () => this.off(handler);
         }

         off(handler: SdkEventHandler): void {
           this.handlers = this.handlers.filter((h) => h !== handler);
         }

         emit(event: SdkEvent): void {
           for (const handler of this.handlers) {
             try { handler(event); } catch { /* subscriber errors don't crash SDK */ }
           }
         }

         removeAll(): void {
           this.handlers = [];
         }
       }
       ```

       ```typescript
       /**
        * Internal folder tree state. Tracks loaded folders, their keys, and metadata.
        * This is the SDK's internal state -- NOT exposed to consumers directly.
        * Consumers get state via events.
        */
       export class FolderTree {
         private folders = new Map<string, FolderState>();

         get(ipnsName: string): FolderState | undefined {
           return this.folders.get(ipnsName);
         }

         set(ipnsName: string, state: FolderState): void {
           this.folders.set(ipnsName, state);
         }

         delete(ipnsName: string): void {
           this.folders.delete(ipnsName);
         }

         has(ipnsName: string): boolean {
           return this.folders.has(ipnsName);
         }

         clear(): void {
           // Clear sensitive key material
           for (const state of this.folders.values()) {
             if (state.folderKey) state.folderKey.fill(0);
             if (state.ipnsKeypair?.privateKey) state.ipnsKeypair.privateKey.fill(0);
           }
           this.folders.clear();
         }

         getAll(): Map<string, FolderState> {
           return new Map(this.folders);
         }
       }
       ```

       ```typescript
       import type { SdkContext, TeeKeys, ProgressCallback, DownloadProgressCallback } from '@cipherbox/sdk-core';
       import * as sdkCore from '@cipherbox/sdk-core';
       import type { FolderChild, FolderMetadata } from '@cipherbox/core';
       import type { CipherBoxClientConfig, FolderState } from './types';
       import { SdkEventEmitter, type SdkEvent, type SdkEventHandler } from './events';
       import { FolderTree } from './state/folder-tree';
       import { KeyCache } from './state/key-cache';

       export class CipherBoxClient {
         private config: CipherBoxClientConfig;
         private ctx: SdkContext;
         private emitter: SdkEventEmitter;
         private folderTree: FolderTree;
         private keyCache: KeyCache;

         constructor(config: CipherBoxClientConfig) {
           this.config = config;
           this.ctx = { apiUrl: config.apiUrl, getAccessToken: config.getAccessToken };
           this.emitter = new SdkEventEmitter();
           this.folderTree = new FolderTree();
           this.keyCache = new KeyCache();
         }

         // Event subscription
         on(handler: SdkEventHandler): () => void { return this.emitter.on(handler); }
         off(handler: SdkEventHandler): void { this.emitter.off(handler); }

         // Lifecycle
         destroy(): void {
           this.folderTree.clear();
           this.keyCache.clear();
           this.emitter.removeAll();
         }

         // ---- Folder operations ----
         async loadFolder(ipnsName: string, folderKey: Uint8Array, ipnsKeypair: {...}): Promise<FolderState> { ... }
         async createFolder(parentIpnsName: string, name: string): Promise<{ ipnsName: string }> { ... }
         async renameItem(folderIpnsName: string, childId: string, newName: string): Promise<void> { ... }
         async moveItem(sourceIpnsName: string, destIpnsName: string, childId: string): Promise<void> { ... }
         async deleteItem(folderIpnsName: string, childId: string): Promise<void> { ... }

         // ---- File operations ----
         async uploadFile(folderIpnsName: string, file: Uint8Array, fileName: string, mimeType: string, onProgress?: ProgressCallback): Promise<{ cid: string }> { ... }
         async downloadFile(cid: string, fileKey: Uint8Array, fileIv: Uint8Array, encryptionMode: string, onProgress?: DownloadProgressCallback): Promise<Uint8Array> { ... }

         // Helper: wrap operation with event emission + error handling
         private async withOperation<T>(name: string, fn: () => Promise<T>): Promise<T> {
           const start = Date.now();
           this.config.onOperationStart?.(name);
           this.emitter.emit({ type: 'operation:start', operation: name });
           try {
             const result = await fn();
             const durationMs = Date.now() - start;
             this.config.onOperationEnd?.(name);
             this.emitter.emit({ type: 'operation:end', operation: name, durationMs });
             return result;
           } catch (error) {
             this.config.onError?.(error as Error);
             this.emitter.emit({ type: 'error', operation: name, error: error as Error });
             throw error;
           }
         }
       }
       ```

       Each public method:
       a) Wraps logic in `withOperation()` for event emission
       b) Reads/updates internal FolderTree state
       c) Calls sdk-core stateless functions with explicit params
       d) Emits folder:loaded/folder:updated/file:uploaded events
       e) Returns result to caller
