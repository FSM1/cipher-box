```typescript
async function deleteAccount(client: LoadClient): Promise<void> {
  const apiBase = await client.page.evaluate(() => {
    return (
      (window as unknown as Record<string, string>).__VITE_API_URL ||
      document.querySelector('meta[name="api-url"]')?.getAttribute('content') ||
      'https://api-staging.cipherbox.cc'
    );
  });
  const deleted = await client.page.evaluate(async (apiUrl: string) => {
    const refreshRes = await fetch(`${apiUrl}/auth/refresh`, {
      method: 'POST',
      credentials: 'include',
    });
    if (!refreshRes.ok) return { ok: false, step: 'refresh', status: refreshRes.status };
    const { accessToken } = await refreshRes.json();
    const deleteRes = await fetch(`${apiUrl}/auth/account`, {
      method: 'DELETE',
      credentials: 'include',
      headers: { Authorization: `Bearer ${accessToken}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ confirmation: 'DELETE' }),
    });
    return { ok: deleteRes.ok, step: 'delete', status: deleteRes.status };
  }, apiBase);
  if (!deleted.ok)
    throw new Error(`Account deletion failed at ${deleted.step}: HTTP ${deleted.status}`);
}
```

1. Discovers the API base URL from the page context using `page.evaluate()`:
   - Check `(window as unknown as Record<string, string>).__VITE_API_URL`
   - Fallback to `document.querySelector('meta[name="api-url"]')?.getAttribute('content')`
   - Final fallback to `'http://localhost:3000'` (local default, NOT staging URL -- this helper is for all environments)

2. Calls `/auth/refresh` (POST, credentials: 'include') to get a fresh access token from the HTTP-only refresh cookie.

3. Calls `/auth/account` (DELETE, credentials: 'include', Authorization: `Bearer ${accessToken}`, body: `{ "confirmation": "DELETE" }`) to delete the account.

4. Wraps the ENTIRE function body in try/catch. On ANY error (network, 401, 500), `console.warn()` the error and return silently. NEVER throw -- this is afterAll teardown and must not mask test failures.

5. Return type: `Promise<void>`.
